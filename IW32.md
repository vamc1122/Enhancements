Definition Name               IWO1_SCREEN_MODIFY


 METHOD if_ex_iwo1_screen_modify~change_screen.
*   DATA :lv_odate  TYPE caufvd-gstrp VALUE '20180101',
*         lv1_odate TYPE date VALUE '20200715'.

     DATA :lv_odate  TYPE caufvd-gstrp VALUE '20141201',
           lv1_odate TYPE date VALUE '20220120'.

   DATA : lt_afvc TYPE TABLE OF afvc,
          ls_afko TYPE afko,
          ls_afvc TYPE afvc.
   TYPES :BEGIN OF ty_afru,
            rueck TYPE afru-rueck,
            rmzhl TYPE afru-rmzhl,
            aueru TYPE afru-aueru,
            aufnr TYPE afru-aufnr,
            vornr TYPE afru-vornr,
          END OF ty_afru.

   DATA : ls_afru TYPE ty_afru,
          lt_afru TYPE TABLE OF ty_afru.

   IF sy-tcode = 'IW32' OR sy-tcode = 'IW31'.
     CLEAR ls_afko.
     SELECT SINGLE * FROM afko INTO ls_afko
          WHERE aufnr EQ is_caufvd-aufnr.
     IF sy-subrc EQ 0.
       CLEAR ls_afvc.
       SELECT  * FROM afvc INTO TABLE lt_afvc
            WHERE aufpl EQ ls_afko-aufpl.
       IF sy-subrc EQ 0.
         CLEAR ls_afru.
         SELECT  rueck rmzhl aueru aufnr vornr FROM afru INTO TABLE lt_afru
           FOR ALL ENTRIES IN lt_afvc
               WHERE  rueck EQ lt_afvc-rueck AND
                      rmzhl EQ lt_afvc-rmzhl AND
                      aueru EQ 'X' AND
                      aufnr EQ is_caufvd-aufnr
               AND vornr EQ is_afvgd-vornr .
         IF sy-subrc NE 0.
           IF is_caufvd-auart EQ 'PM03' AND is_caufvd-ilart NE '302' .
             IF  is_caufvd-gstrp GE lv_odate  AND is_caufvd-gstrp LT lv1_odate.
               IF i_screen_name = 'AFVGD-ARBEI'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-ARBEH'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-DAUNE'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-INDET'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
             ELSE.
               IF i_screen_name = 'AFVGD-ARBEI'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-ARBEH'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-DAUNE'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-INDET'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
             ENDIF.
           ELSEIF is_caufvd-auart NE 'PM03' AND           "Exclude Order type 'PM03' and activity '302'
            is_caufvd-ilart NE '302'.
             IF  is_caufvd-gstrp GE lv_odate  AND is_caufvd-gstrp LT lv1_odate.
               IF i_screen_name = 'AFVGD-ARBEI'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-ARBEH'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-DAUNE'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-INDET'.
                 screen-input = 1.
                 MODIFY SCREEN.
               ENDIF.
             ELSE.
               IF i_screen_name = 'AFVGD-ARBEI'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-ARBEH'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-DAUNE'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
               IF i_screen_name EQ 'AFVGD-INDET'.
                 screen-input = 0.
                 MODIFY SCREEN.
               ENDIF.
             ENDIF.
           ENDIF.
         ENDIF.
       ENDIF.
     ENDIF.
   ENDIF.
 ENDMETHOD.
