  METHOD if_ex_badi_material_check~check_data.
*---------------------------------------------------------------------*
*       Default the Purchase key based on Material Type               *
*                                                                     *
*---------------------------------------------------------------------*
*  TR                    :  IN1K911585
*  Technical Consultant  :  Chetan
*  Functional Consultant :  Sunil Sarangamath

*    IF sy-tcode = 'MM01' OR sy-tcode = ''." OR sy-tcode = 'ZMDM_MRO'.
    DATA: ls_data TYPE zxx_generic_data.
    DATA: ls_data1 TYPE zxx_generic_data.
    DATA ls_mara TYPE mara.
    DATA ls_t006 TYPE t006.
    CLEAR: ls_data.

    SELECT SINGLE *
      FROM mara
      INTO ls_mara
      WHERE matnr EQ wmara-matnr.
    IF sy-subrc NE 0.

*  ENDIF.

      "Get the Material Type is activated for Tolerance default
      SELECT SINGLE * FROM
                      zxx_generic_data
                      INTO ls_data
                      WHERE idnt1 = 'MATERIAL'
                      AND   idnt2 = 'TOLERANCE'
                      AND   idnt3 = 'MTART'
                      AND   flval = wmara-mtart.
      IF sy-subrc EQ 0.
        "If Uom Type dimensions are – MASS , VOLUME, & LENGTH
        IF wmara-meins IS NOT INITIAL.
          SELECT SINGLE *
            FROM t006
            INTO ls_t006
            WHERE msehi = wmara-meins
              AND dimid IN ('MASS' , 'LENGTH' ,'VOLUME' ).
          IF sy-subrc EQ 0.

            "Check for the key maintained in generic table
            CLEAR: ls_data1.
            SELECT SINGLE * FROM
                            zxx_generic_data
                            INTO ls_data1
                            WHERE idnt1 = 'MATERIAL'
                            AND   idnt2 = 'TOLERANCE'
                            AND   idnt3 = 'KEY'.
            IF sy-subrc EQ 0 .
*                      AND   flval = wmara-mtart.
              MOVE ls_data1-flval TO wmara-ekwsl.
              MOVE ls_data1-flval TO cmara-ekwsl.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.

    ENDIF.
  ENDMETHOD.
